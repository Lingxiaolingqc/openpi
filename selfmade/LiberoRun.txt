2.客户端
cd /home/data/xiaoqinchuan/projects/openpi

source /home/data/xiaoqinchuan/envs/openpi-libero/bin/activate

export SUITE=libero_spatial
export TRIALS=2
export RUN_DIR="/home/data/xiaoqinchuan/results/libero/${SUITE}-${TRIALS}-seed7"

export PYTHONPATH="$PWD/third_party/libero:$PYTHONPATH"
export LIBERO_CONFIG_PATH=/home/data/xiaoqinchuan/config/libero

export CUDA_VISIBLE_DEVICES=1
export MUJOCO_GL=egl
export MUJOCO_EGL_DEVICE_ID=1

set -o pipefail

python examples/libero/main.py \
  --args.host localhost \
  --args.port 8000 \
  --args.task-suite-name "$SUITE" \
  --args.num-trials-per-task "$TRIALS" \
  --args.seed 7 \
  --args.resize-size 224 \
  --args.replan-steps 5 \
  --args.num-steps-wait 10 \
  --args.video-out-path "$RUN_DIR/videos" \
  2>&1 | tee "$RUN_DIR/client.log"

client_exit_code=${PIPESTATUS[0]}
echo "LIBERO client exit code: $client_exit_code" | tee -a "$RUN_DIR/client.log"
